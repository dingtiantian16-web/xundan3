const $ = selector => document.querySelector(selector);

const escapeHtml = value =>
  String(value ?? "").replace(/[&<>"']/g, char => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    '"': "&quot;",
    "'": "&#39;"
  })[char]);

let rawLog = "";
let rows = [];

const fileInput = $("#file");
const dropArea = $("#drop");

fileInput.addEventListener("change", () => {
  loadFile(fileInput.files[0]);
});

["dragenter", "dragover"].forEach(eventName => {
  dropArea.addEventListener(eventName, event => {
    event.preventDefault();
    dropArea.classList.add("drag");
  });
});

["dragleave", "drop"].forEach(eventName => {
  dropArea.addEventListener(eventName, event => {
    event.preventDefault();
    dropArea.classList.remove("drag");
  });
});

dropArea.addEventListener("drop", event => {
  loadFile(event.dataTransfer.files[0]);

function loadFile(file) {
  if (!file) return;

  const reader = new FileReader();

  reader.onload = () => {
    rawLog = reader.result;
    $("#fileName").textContent = file.name;
  };

  reader.readAsText(file, "utf-8");
}

function parseMessages(block) {
  const messages = [];
  const lines = block.split(/\r?\n/);
  let currentMessage;

  for (let line of lines) {
    line = line.trim();

    const header = line.match(
      /^(.+?)\s+(20\d{2}-\d{2}-\d{2})\s+(\d{2}:\d{2}:\d{2})$/
    );

    if (header) {
      if (currentMessage) messages.push(currentMessage);

      currentMessage = {
        speaker: header[1],
        date: header[2],
        time: header[3],
        body: []
      };
    } else if (
      currentMessage &&
      line &&
      !line.startsWith("/***")
    ) {
      currentMessage.body.push(
        line
          .replace(/<br\s*\/?\s*>/gi, " ")
          .replace(/<[^>]*>/g, "")
      );
    }
  }

  if (currentMessage) messages.push(currentMessage);
  return messages;
}

function matchCustomerId(sourceId, completeIds) {
  const candidates = completeIds.filter(id => {
    if (!sourceId.includes("*")) return id === sourceId;

    const prefix = sourceId.slice(0, sourceId.indexOf("*"));
    const suffix = sourceId.slice(sourceId.lastIndexOf("*") + 1);

    return id.startsWith(prefix) && id.endsWith(suffix);
  });

  if (candidates.length === 1) return candidates[0];

  if (candidates.length > 1) {
    return "待核对：" + candidates.join("、");
  }

  return sourceId || "缺失";
}

const categoryRules = [
  ["价格异议", /贵|便宜|价格|优惠/],
  ["售后保障", /售后|质保|换新|补发|包赔|贴坏/],
  ["贴膜操作", /怎么贴|教程|安装|无尘仓|贴膜神器|视频/],
  ["贴后问题", /气泡|白边|进灰|翘边|瑕疵|不清楚|碎了/],
  ["机型适配", /型号|适配|通用|iphone|苹果|\b1[4-9](?:p|pro|pm|air|plus)/i],
  ["产品差异", /区别|不同|原感|无界|泰坦|同款|一样/],
  ["工艺参数", /热弯|3d|2\.5d|反射|透光|ar|黑边|防窥|厚度|材质/i],
  ["库存物流", /发货|物流|仓库|到货/],
  ["订单咨询", /订单|地址/]
];

function analyze() {
  if (!rawLog) {
    alert("请先选择聊天记录");
    return;
  }

  const completeIds = $("#ids").value
    .split(/\r?\n/)
    .map(value => value.trim())
    .filter(Boolean);

  const sessionBlocks = rawLog
    .split(/\/\*+以下为一通会话\*+\//)
    .filter(block => block.includes("会话结束"));

  rows = sessionBlocks.map((block, index) => {
    const messages = parseMessages(block);

    const customerMessages = messages.filter(
      message => !message.speaker.startsWith("图拉斯旗舰店-")
    );

    const serviceMessages = messages.filter(
      message => message.speaker.startsWith("图拉斯旗舰店-")
    );

    const sourceId = customerMessages[0]?.speaker || "";

    const customerQuestion =
      customerMessages
        .flatMap(message => message.body)
        .filter(text =>
          text &&
          !/^https?:\/\//.test(text) &&
          !/^#E-/.test(text)
        )
        .join("；") ||
      "无可识别文字问题／仅图片或链接";

    const serviceAnswer = serviceMessages
      .flatMap(message => message.body)
      .join("；");

    const orderConflict =
      /订单已确认|已收到你的订单/.test(serviceAnswer);

    const isAfterSale =
      /咨询订单号|气泡|进灰|碎了|昨天贴|刚贴|瑕疵|贴上|贴完|不清楚|可以售后/
        .test(customerQuestion);

    const summary =
      categoryRules
        .filter(rule => rule[1].test(customerQuestion))
        .map(rule => rule[0])
        .join("、") ||
      "其他咨询";

    let risk =
      "原因不明确：未下单事实已确认，但不能从沉默推断具体原因";

    let optimization = "证据不足";

    let evidence =
      `客户：${customerQuestion.slice(0, 140)}；未见明确拒购原因`;

    let advice =
      "确认机型与选择标准，补足未解决信息；客户沉默后进行一次自然跟进。";

    if (/在优惠个5块就买了/.test(customerQuestion)) {
      risk = "明确购买条件未满足：希望再优惠5元";
      optimization = "价格异议承接";
      evidence =
        "客户明确表示“再优惠5块就买了”，客服继续重复活动说明";
      advice =
        "核对可用优惠；无法满足时明确说明，并提供预算内替代款。";
    } else if (
      /17pro/i.test(customerQuestion) &&
      /17promax/i.test(serviceAnswer)
    ) {
      risk = "可能阻碍：机型回复错配，增加买错顾虑";
      optimization = "机型/SKU核对";
      evidence = "客户说17 Pro，回复涉及17 Pro Max";
      advice =
        "复述完整机型，核对SKU后只发送一个匹配链接。";
    } else if (/我不要防窥膜/.test(customerQuestion)) {
      risk =
        "可能阻碍：推荐款式错配；后续虽纠正，仍可能增加决策成本";
      optimization = "需求确认";
      evidence = "客户明确表示不要防窥膜";
      advice =
        "发链接前先确认机型及高清/防窥需求，纠正时说明旧链接不用选。";
    } else if (/贴膜神器是一次性/.test(customerQuestion)) {
      risk = "可能阻碍：工具复用问题未被准确回答";
      optimization = "精准回答";
      evidence = "客户问工具，客服回复内容偏向胶质拆卸";
      advice =
        "区分膜和无尘仓，核实工具能否复用及耗材限制后直答。";
    } else if (/就一张，贴坏了咋办/.test(customerQuestion)) {
      risk = "可能阻碍：单片贴坏顾虑未明确消除";
      optimization = "保障说明";
      evidence =
        "客服使用“不会贴坏”等保证性表达，未先明确适用条件";
      advice =
        "说明片数、工具、教程和该SKU实际保障条件，不保证零失败。";
    } else if (isAfterSale) {
      risk = "售后服务问题，不计为本次售前购买阻碍";
      optimization = "售后闭环";
      advice =
        "收集图片、订单与机型，说明处理步骤并确认结果，不继续催拍。";
    } else if (/转接/.test(serviceAnswer)) {
      risk = "可能阻碍：本通未闭环，转接结果未知";
      optimization = "转接闭环";
      advice =
        "同步客户需求，告知下一步，并结合后续会话确认是否解决。";
    }

    if (orderConflict) {
      evidence +=
        "；日志出现订单提醒，与“本批未下单”口径冲突，不计成交";
    }

    return {
      number: index + 1,
      date: customerMessages[0]?.date || messages[0]?.date || "",
      customerId: matchCustomerId(sourceId, completeIds),
      sourceId,
      stage: isAfterSale ? "售后" : "售前",
      status: "未下单",
      question: customerQuestion,
      summary,
      risk,
      evidence,
      optimization,
      advice,
      orderConflict
    };
  });

  renderResults();
}

function renderResults() {
  const preSaleCount =
    rows.filter(row => row.stage === "售前").length;

  $("#sSessions").textContent = rows.length;
  $("#sPre").textContent = preSaleCount;
  $("#sAfter").textContent = rows.length - preSaleCount;

  $("#sConflict").textContent =
    rows.filter(row => row.orderConflict).length;

  $("#detailBody").innerHTML = rows.map(row => `
    <tr>
      <td>${escapeHtml(row.date)}</td>
      <td>${escapeHtml(row.customerId)}</td>
      <td><span class="tag">${row.stage}</span></td>
      <td>未下单</td>
      <td>${escapeHtml(row.question)}</td>
      <td>${escapeHtml(row.summary)}</td>
    </tr>
  `).join("");

  $("#lossBody").innerHTML = rows.map(row => `
    <tr>
      <td>${escapeHtml(row.customerId)}</td>
      <td>
        ${escapeHtml(row.date)} · 会话${row.number}
        <br><small>${escapeHtml(row.sourceId)}</small>
      </td>
      <td>${escapeHtml(row.risk)}</td>
      <td>${escapeHtml(row.evidence)}</td>
      <td>${escapeHtml(row.optimization)}</td>
      <td>${escapeHtml(row.advice)}</td>
    </tr>
  `).join("");

  const frequencyRules = [
    ["款式区别/如何选择", /区别|不同|同款|一样|ar膜就这一款/],
    ["黑边/覆盖/边缘工艺", /黑边|覆盖|全屏|弧边|弧度|3d|2\.5d|热弯|小了一点|短一截/i],
    ["贴膜操作/工具/贴坏顾虑", /怎么贴|安装视频|使用视频|贴膜视频|贴膜神器|贴坏|包赔|手残/],
    ["保障/质保/换新", /质保|换新|给换|包赔|贴坏|质量问题.*换/],
    ["机型/型号/链接适配", /16pm|15plus|17 promax|17pro|18pro|17 18|什么型号/i],
    ["材质/厚度", /康宁|厚度|钢化膜，还是软/],
    ["价格/优惠", /贵|优惠|便宜/]
  ];

  const frequencies = frequencyRules.map(rule => [
    rule[0],
    rows.filter(row =>
      !row.question.includes("小冰块") &&
      rule[1].test(row.question)
    ).length
  ]).sort((a, b) => b[1] - a[1]);

  const maximum = frequencies[0]?.[1] || 1;

  $("#freqList").innerHTML = frequencies.map(item => `
    <div class="freq-item">
      <span>${item[0]}</span>
      <div class="bar">
        <i style="width:${item[1] / maximum * 100}%"></i>
      </div>
      <b>${item[1]}</b>
    </div>
  `).join("");

  $("#shareText").value = buildGroupShare(frequencies);

  $("#empty").hidden = true;
  $("#results").hidden = false;
}

function buildGroupShare(frequencies) {
  return `家人们，近日询单总结来啦～[冲！]

      rows.map(row => [

      ])
    ],
    [
      "流失分析",
      ["客户ID","会话定位","购买阻碍/流失风险","判断依据","可优化项","优化建议"],
      rows.map(row => [
        row.customerId,
        `${row.date} 会话${row.number}`,
        row.risk,
        row.evidence,
        row.optimization,
        row.advice
      ])
    ]
  ];

  const worksheets = sheets.map(sheet => `
    <Worksheet ss:Name="${sheet[0]}">
      <Table>
        ${[sheet[1], ...sheet[2]].map((row, index) => `
          <Row>
            ${row.map(cell => `
              <Cell${index === 0 ? ' ss:StyleID="header"' : ""}>
                <Data ss:Type="String">${escapeHtml(cell)}</Data>
              </Cell>
            `).join("")}
          </Row>
        `).join("")}
      </Table>
    </Worksheet>
  `).join("");

  const workbook = `<?xml version="1.0"?>
    <Workbook
      xmlns="urn:schemas-microsoft-com:office:spreadsheet"
      xmlns:ss="urn:schemas-microsoft-com:office:spreadsheet">
      <Styles>
        <Style ss:ID="header">
          <Font ss:Bold="1"/>
          <Interior ss:Color="#DFF8F3" ss:Pattern="Solid"/>
        </Style>
      </Styles>
      ${worksheets}
    </Workbook>`;

  const blob = new Blob(
    [workbook],
    { type: "application/vnd.ms-excel" }
  );

  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = "无界AR膜询单及流失分析.xls";
  link.click();

  URL.revokeObjectURL(link.href);
});
